\chapter{Evaluation of $\|\be\|_{\hdo}$ for the General Taylor Green Vortex}
\label{app:tg:seminorm:general}

The general form of the 3D Taylor Green Vortex on the 3D torus $\Omega$ with characteristic length $L$ is given in (\ref{eq:TGV})
To simplify the notation we drop the subscript $TGV$ and substitute in 
$$x' = 2\pi x/L, \quad y' = 2\pi y/L, \quad z' = 2\pi z/L.$$
The formula for $\|\be\|_{\hdo}$ is
$$\|\be\|_{\hdo} = \|\nabla\be\|_{L^2} = \left[ \sum_{i,j=1}^3 \int_{\Omega} (\partial_i\be_j)^2 \,d\x \right]^{1/2},$$
where $\partial_i$ is the partial derivative with respect to the i-th component and $\be_j$ is the j-th component of $\be$. Because the 3rd component of $\be$ is zero, the partial derivative is also zero. This leaves us with six integrals to evaluate.
\begin{align*}
\sum_{i,j=1}^3 \int_{\Omega} (\partial_i\be_j)^2 \,d\x &= \int_{\Omega} (\partial_1\be_1)^2 \, d\x + \int_{\Omega} (\partial_2\be_1)^2 \, d\x + \int_{\Omega} (\partial_3\be_1)^2 \, d\x \\
&+ \int_{\Omega} (\partial_1\be_2)^2 \, d\x + \int_{\Omega} (\partial_2\be_2)^2 \, d\x + \int_{\Omega} (\partial_3\be_2)^2 \, d\x.
\end{align*}
Evaluating each partial derivative
\begin{small}
\begin{align*}
\sum_{i,j=1}^3 \int_{\Omega} (\partial_i\be_j)^2 \,d\x &= \int_{\Omega} (2\pi V_0/L \cos(x')\cos(y')\cos(z'))^2 \, d\x \\ &+ \int_{\Omega} (-2\pi V_0/L \sin(x')\sin(y')\cos(z'))^2 \, d\x \\
&+ \int_{\Omega} (-2\pi V_0/L \sin(x')\cos(y')\sin(z'))^2 \, d\x \\ &+ \int_{\Omega} (2\pi V_0/L \sin(x')\sin(y')\cos(z'))^2 \, d\x \\ 
&+ \int_{\Omega} (-2\pi V_0/L \cos(x')\cos(y')\cos(z'))^2 \, d\x \\ &+ \int_{\Omega} (2\pi V_0/L \cos(x')\sin(y')\sin(z'))^2 \, d\x,
\end{align*}
\end{small}
which simplifies to
\begin{small}
\begin{align*}
\sum_{i,j=1}^3 \int_{\Omega} (\partial_i\be_j)^2 \,d\x &= 4\pi^2 V_0^2/L^2\int_{\Omega} \cos^2(x')\cos^2(y')\cos^2(z') \, d\x \\ &+ 4\pi^2 V_0^2/L^2\int_{\Omega} \sin^2(x')\sin^2(y')\cos^2(z') \, d\x \\
&+ 4\pi^2 V_0^2/L^2\int_{\Omega} \sin^2(x')\cos^2(y')\sin^2(z') \, d\x \\ &+ 4\pi^2 V_0^2/L^2\int_{\Omega} \sin^2(x')\sin^2(y')\cos^2(z') \, d\x \\ 
&+ 4\pi^2 V_0^2/L^2\int_{\Omega} \cos^2(x')\cos^2(y')\cos^2(z') \, d\x \\ &+ 4\pi^2 V_0^2/L^2\int_{\Omega} \cos^2(x')\sin^2(y')\sin^2(z') \, d\x.
\end{align*}
\end{small}
Because the integrand is separable, the integrals can be reduced to products of,
\begin{equation*}
\int_{0}^L \cos^2(x')\, dx = \frac{L}{2}, \qquad \text{and} \qquad \int_{0}^L \sin^2(x')\, dx = \frac{L}{2}.
\end{equation*}
Therefore, each of the six integrals is equal to 
\begin{equation*}
\frac{4\pi^2 V_0^2}{L^2}\frac{L^3}{8}.
\end{equation*}
The sum of the six integrals is
$$\sum_{i,j=1}^3 \int_{\Omega} (\partial_i\be_j)^2 \,d\x = 6\frac{4\pi^2V_0^2}{L^2}\frac{L^3}{8} = 3\pi^2V_0^2L. $$
Substituting into the formula for $\|\cdot\|_{\hdo}$ gives
\\
$$\|\be_{TGV}\|_{\hdo} = \left[ 3\pi^2V_0^2L \right]^{1/2} = \sqrt{3L}\pi V_0.$$
