\chapter{Evaluation of $\|\nabla\times\be\|_{L^\infty}$ for the General Taylor Green Vortex}
\label{app:tg:vorticity:general}

The general form of the 3D Taylor Green Vortex on the 3D torus $\Omega$ with characteristic length $L$ is given in (\ref{eq:TGV}). To simplify the notation we drop the subscript $TGV$ and substitute in
$$x' = 2\pi x/L, \quad y' = 2\pi y/L, \quad z' = 2\pi z/L.$$
The vorticity of $\be$ is
$$\nabla \times \be = \begin{bmatrix}
           \partial_2\be_3 - \partial_3\be_2 \\
           \partial_3\be_1 - \partial_1\be_3 \\
           \partial_1\be_2 - \partial_2\be_1.
         \end{bmatrix}$$
where $\partial_i$ is the partial derivative with respect to the i-th component and $\be_j$ is the j-th component of $\be$. Substituting in the partial derivatives
\\
\begin{small}
$$\nabla \times \be = \begin{bmatrix}
           0 - (-(2\pi V_0/L)\cos(x')\sin(y')\sin(z') \\
           -(2\pi V_0/L) \sin(x')\cos(y')\sin(z') - 0\\
           (2\pi V_0/L)\sin(x')\sin(y')\cos(z') + (2\pi V_0)\sin(x')\sin(y')\cos(z'),
         \end{bmatrix}$$
\end{small}
which simplifies to 
\begin{small}
$$\nabla \times \eta_{TG} = \begin{bmatrix}
           (2\pi V_0/L) \cos(x')\sin(y')\sin(z') \\
           -(2\pi V_0/L) \sin(x')\cos(y')\sin(z')\\
           (4\pi V_0/L)\sin(x')\sin(y')\cos(z')
         \end{bmatrix}.$$
\end{small}
The $\|\nabla \times \be\|_{L^{\infty}}$ norm is the maximum absolute value of the components of $\nabla \times \be$. 
\begin{align*}
\|\nabla \times \be_{TGV}\|_{L^{\infty}} &= \max( \vert(2\pi V_0/L) \cos(x')\sin(y')\sin(z')\vert \\
&, \vert-(2\pi V_0/L) \sin(x')\cos(y')\sin(z')\vert \\
&, \vert(4\pi V_0/L)\sin(x')\sin(y')\cos(z')\vert)
\end{align*}
The maximums are
\begin{equation*}
\|\nabla \times \be_{TGV}\|_{L^{\infty}} = \max( (2\pi V_0/L), (2\pi V_0/L), (4\pi V_0/L))
\end{equation*}
Therefore,
$$\|\nabla \times \be_{TGV}\|_{L^{\infty}} = 4\pi V_0/L.$$
