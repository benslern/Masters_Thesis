\section*{Notation}
\begin{description}[font=\rmfamily\bfseries, leftmargin=3cm, style=nextline]
	\item[\textnormal{x}] Plain-text to denote scalar x
	\item[$\bds x$] Bold-text to denote vector x
	\item[$\bds x^\intercal$] Transpose of vector x
	\item[$I$] Identity Operator
\end{description}
